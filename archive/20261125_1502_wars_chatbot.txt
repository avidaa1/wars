function decryptBlob(blob,passphrase,onURL){

  const salt=Uint8Array.from(atob(parts[1]),c=>c.charCodeAt(0));
  const iv=Uint8Array.from(atob(parts[2]),c=>c.charCodeAt(0));
  const ctRaw=Uint8Array.from(atob(parts[3]),c=>c.charCodeAt(0));
  const data=ctRaw.slice(0,ctRaw.length-16),tag=ctRaw.slice(ctRaw.length-16);
  const combined=new Uint8Array(data.length+tag.length);combined.set(data);combined.set(tag,data.length);
  const enc=new TextEncoder();
  return crypto.subtle.importKey('raw',enc.encode(passphrase),{name:'PBKDF2'},false,['deriveKey'])
    .then(k=>crypto.subtle.deriveKey({name:'PBKDF2',salt,iterations:100000,hash:'SHA-256'},k,{name:'AES-GCM',length:256},false,['decrypt']))

}

function lev(a,b){
  if(Math.abs(a.length-b.length)>2)return 9;
  const m=a.length,n=b.length,d=Array.from({length:m+1},(_,i)=>[i,...Array(n).fill(0)]);
  for(let j=0;j<=n;j++)d[0][j]=j;
  for(let i=1;i<=m;i++)for(let j=1;j<=n;j++)
    d[i][j]=Math.min(d[i-1][j]+1,d[i][j-1]+1,d[i-1][j-1]+(a[i-1]===b[j-1]?0:1));
  return d[m][n];
}
function detectTone(msg){
  const m=msg.toLowerCase();
  if(/[!]{1,}|wow|amazing|cool|awesome|insane/.test(m))return'excited';
  if(/\?$|how|why|what|explain|works|technical/.test(m))return'curious';
  if(/really|prove|sure|doubt|fake|nonsense|why should/.test(m))return'skeptical';
  if(/\b(hey|sup|yo|lol|gimme)\b/.test(m))return'casual';
  return'neutral';
}

function scoreEntry(e,low,words){
  let sc=0;
  e.k.forEach(k=>{
    if(low===k||words.includes(k))sc+=10;
    else if(low.includes(k)&&k.length>=4)sc+=k.split(' ').length*k.length*2;
    else if(k.startsWith(low)||low.startsWith(k))sc+=Math.min(k.length,low.length)*1.2;
    else words.forEach(w=>{
      if(w.length>=3&&k.startsWith(w))sc+=w.length*0.9;
      else if(w.length>=4&&k.includes(w))sc+=w.length*0.5;
      else if(w.length>=6&&k.length>=6&&lev(w,k)<=1)sc+=w.length*0.6;
    });
  });
  return sc;
}
